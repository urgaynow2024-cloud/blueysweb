export default function ContactCommissionForm() {
  const [step, setStep] = useState(0);
  const [submitted, setSubmitted] = useState(false);
  const [error, setError] = useState(false);
  const [submitting, setSubmitting] = useState(false);
  const [formError, setFormError] = useState<string | null>(null);

  const [name, setName] = useState("");
  const [discord, setDiscord] = useState("");
  const [email, setEmail] = useState("");
  const [description, setDescription] = useState("");
  const [budget, setBudget] = useState("");
  const [deadline, setDeadline] = useState("");
  const [notes, setNotes] = useState("");
  const [references, setReferences] = useState("");
  const [agreed, setAgreed] = useState(false);
  const [ownsAssets, setOwnsAssets] = useState(false);
  const [adoptableProof, setAdoptableProof] = useState(false);
  const [refundsUnderstood, setRefundsUnderstood] = useState(false);

  const [uploads, setUploads] = useState<UploadItem[]>([]);
  const [uploading, setUploading] = useState(false);
  const [uploadError, setUploadError] = useState<string | null>(null);

  const nameId = useId();
  const discordId = useId();
  const descId = useId();
  const budgetId = useId();
  const deadlineId = useId();
  const refsId = useId();
  const notesId = useId();

  const canNext = () => {
    if (step === 0) return name.trim().length > 0 && discord.trim().length > 0;
    if (step === 1) return description.trim().length >= 10;
    if (step === 2) return true;
    if (step === 3) return agreed && ownsAssets && adoptableProof && refundsUnderstood;
    return false;
  };

  const handleNext = () => { if (canNext()) setStep((s) => Math.min(s + 1, 3)); };
  const handleBack = () => { if (step > 0) setStep((s) => Math.max(s - 1, 0)); };

  const addFiles = useCallback((files: File[]) => {
    setUploadError(null);
    const valid = files.filter((f) => {
      const ok = f.type.startsWith("image/") || f.type.startsWith("video/") || f.type === "application/pdf";
      if (!ok) setUploadError("Only image, video, and PDF files are supported.");
      return ok;
    });
    if (valid.length === 0) return;
    setUploads((prev) => [
      ...prev,
      ...valid.map((file) => ({
        file,
        previewUrl: file.type.startsWith("image/") ? URL.createObjectURL(file) : "",
        status: "pending" as const,
        progress: 0,
      })),
    ]);
  }, []);

  const removeUpload = useCallback((index: number) => {
    setUploads((prev) => {
      const item = prev[index];
      if (item?.previewUrl) URL.revokeObjectURL(item.previewUrl);
      return prev.filter((_, i) => i !== index);
    });
  }, []);

  const uploadAll = async () => {
    if (uploads.length === 0) return [] as string[];
    setUploading(true);
    setUploadError(null);
    const urls: string[] = [];
    for (let i = 0; i < uploads.length; i++) {
      const item = uploads[i];
      if (item.status === "success" && item.result) {
        urls.push(item.result.url);
        continue;
      }
      setUploads((prev) => prev.map((u, idx) => (idx === i ? { ...u, status: "uploading", progress: 0 } : u)));
      try {
        const result = await uploadMedia(item.file, "commission-reference", undefined, (p) => {
          setUploads((prev) => prev.map((u, idx) => (idx === i ? { ...u, progress: p.percentage } : u)));
        });
        setUploads((prev) => prev.map((u, idx) => (idx === i ? { ...u, status: "success", progress: 100, result } : u)));
        urls.push(result.url);
      } catch (err) {
        const message = err instanceof UploadError ? err.message : "Upload failed";
        setUploads((prev) => prev.map((u, idx) => (idx === i ? { ...u, status: "error", error: message } : u)));
        setUploadError(message);
      }
    }
    setUploading(false);
    return urls;
  };

  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
    e.preventDefault();
    setError(false);
    setFormError(null);

    if (!agreed || !ownsAssets || !adoptableProof || !refundsUnderstood) {
      setError(true);
      setFormError("Please read and agree to all terms below (Terms of Service, asset ownership, adoptable proof, and refund policy) before submitting.");
      setStep(3);
      return;
    }

    const urls = await uploadAll();
    if (uploadError && urls.length === 0) {
      setError(true);
      setFormError(uploadError);
      return;
    }

    const data = {
      name: name.trim(),
      discord: discord.trim(),
      email: email.trim() || undefined,
      description: description.trim(),
      budget: budget.trim() || undefined,
      deadline: deadline.trim() || undefined,
      references: urls.length ? urls.join("\n") : (references.trim() || undefined),
      notes: notes.trim() || undefined,
      agreed,
      owns_assets: ownsAssets,
      adoptable_proof: adoptableProof,
      refunds_understood: refundsUnderstood,
    };

    try {
      setSubmitting(true);
      const res = await fetch("/api/commission", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(data),
      });
      if (res.ok) {
        setSubmitted(true);
      } else {
        setError(true);
        setFormError("Failed to send request. Please try again or contact me directly on Discord.");
      }
    } catch {
      setError(true);
      setFormError("Network error. Please check your connection and try again.");
    } finally {
      setSubmitting(false);
    }
  };
